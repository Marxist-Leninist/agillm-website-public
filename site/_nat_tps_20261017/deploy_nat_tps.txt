MARKER = 'NAT_LIVE_TPS_20261017'
EXPECTED_MARKERS = 4

def sha256_of(path):
    h = hashlib.sha256()
    with open(path, 'rb') as f:
        for chunk in iter(lambda: f.read(1 << 20), b''):
            h.update(chunk)
    return h.hexdigest()

def main():
    if len(sys.argv) < 2:
        print(__doc__); return 2
    path = sys.argv[1]
    check_only = '--check' in sys.argv
    raw = io.open(path, 'rb').read()
    sha_before = hashlib.sha256(raw).hexdigest()
    src = raw.decode('utf-8')
    if src.count(MARKER):
        print(f'ALREADY APPLIED: {src.count(MARKER)} markers present in {path}; nothing to do'); return 0
    for i, (old, new) in enumerate(EDITS, 1):
        n = src.count(old)
        if n != 1:
            print(f'ABORT: edit {i} matched {n} times (expected 1); file has drifted; nothing written'); return 4
        src = src.replace(old, new)
    out = src.encode('utf-8')
    assert src.count(MARKER) == EXPECTED_MARKERS, src.count(MARKER)
    print(f'before sha256 {sha_before} bytes {len(raw)}')
    print(f'after  sha256 {hashlib.sha256(out).hexdigest()} bytes {len(out)}')
    if check_only:
        print('CHECK OK: all edits apply exactly once; not written'); return 0
    stamp = time.strftime('%Y%m%dT%H%M%SZ', time.gmtime())
    backup = f'{path}.bak-nat-tps-{stamp}'
    tmp = f'{path}.nat-tps.{stamp}.tmp'
    shutil.copy2(path, backup)
    st = os.stat(path)
    with open(tmp, 'wb') as f:
        f.write(out); f.flush(); os.fsync(f.fileno())
    os.chmod(tmp, st.st_mode & 0o7777)
    # compare-and-swap: refuse if another writer changed the file since it was read
    if sha256_of(path) != sha_before:
        os.unlink(tmp)
        print('ABORT: file changed during patching (sha mismatch); nothing written; backup left at', backup); return 3
    os.replace(tmp, path)
    dfd = os.open(os.path.dirname(path) or '.', os.O_RDONLY)
    try: os.fsync(dfd)
    finally: os.close(dfd)
    print(f'WRITTEN {path}; backup {backup}; live sha256 {sha256_of(path)}')
    return 0

if __name__ == '__main__':
    sys.exit(main())
